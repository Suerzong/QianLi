cmake_minimum_required(VERSION 3.8)
project(qianli_navigation)
find_package(ament_cmake REQUIRED)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/navigation_test.py DESTINATION lib/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})

  ament_add_test(exploration_envelope COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test/test_exploration_envelope.py"
    GENERATE_RESULT_FOR_RETURN_CODE_ZERO TIMEOUT 60)
endif()
ament_package()
